cmake_minimum_required(VERSION 3.22)
project(RootsDesktopSurface LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The desktop capture adapter requires Windows.")
endif()
option(ROOTS_SURFACE_OCR "Build Windows Graphics Capture and Windows OCR lane" ON)
add_executable(roots-surface launcher/main.cpp)
target_compile_features(roots-surface PRIVATE cxx_std_20)
target_compile_definitions(roots-surface PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(roots-surface PRIVATE ole32 oleaut32 uiautomationcore user32 wtsapi32)
if(ROOTS_SURFACE_OCR)
  target_sources(roots-surface PRIVATE acquisition/ocr.cpp acquisition/tesseract.cpp)
  target_compile_definitions(roots-surface PRIVATE ROOTS_SURFACE_OCR=1)
  target_link_libraries(roots-surface PRIVATE d3d11 dxgi dwmapi runtimeobject windowsapp)
endif()
if(MSVC)
  target_compile_options(roots-surface PRIVATE /W4 /EHsc /utf-8)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(roots-surface-pixels-check tests/ocr_pixels_checks.cpp)
  target_include_directories(roots-surface-pixels-check PRIVATE acquisition)
  target_compile_features(roots-surface-pixels-check PRIVATE cxx_std_20)
  add_test(NAME roots-surface-pixels-check COMMAND roots-surface-pixels-check)
endif()
