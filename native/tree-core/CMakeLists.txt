cmake_minimum_required(VERSION 3.22.1)
project(roots_tree_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Android's platform sqlite is not a public NDK dependency. Statically link
# the pinned upstream amalgamation, or an explicitly supplied source directory.
set(ROOTS_SQLITE_AMALGAMATION_DIR "" CACHE PATH "Directory containing sqlite3.c and sqlite3.h")
option(ROOTS_TREE_PROVISION_SQLITE "Use the existing pinned SQLite provisioner on desktop" OFF)
if((ANDROID OR ROOTS_TREE_PROVISION_SQLITE) AND NOT ROOTS_SQLITE_AMALGAMATION_DIR)
    include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/ProvisionSQLite.cmake")
    roots_provision_sqlite(ROOTS_SQLITE_AMALGAMATION_DIR)
endif()
if(ROOTS_SQLITE_AMALGAMATION_DIR)
    if(NOT EXISTS "${ROOTS_SQLITE_AMALGAMATION_DIR}/sqlite3.c" OR NOT EXISTS "${ROOTS_SQLITE_AMALGAMATION_DIR}/sqlite3.h")
        message(FATAL_ERROR "SQLite source directory must contain sqlite3.c and sqlite3.h: ${ROOTS_SQLITE_AMALGAMATION_DIR}")
    endif()
    add_library(roots_tree_sqlite STATIC "${ROOTS_SQLITE_AMALGAMATION_DIR}/sqlite3.c")
    target_include_directories(roots_tree_sqlite PUBLIC "${ROOTS_SQLITE_AMALGAMATION_DIR}")
    target_compile_definitions(roots_tree_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
    set_target_properties(roots_tree_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
    if(UNIX AND NOT ANDROID)
        find_package(Threads REQUIRED)
        target_link_libraries(roots_tree_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
    set(ROOTS_SQLITE_TARGET roots_tree_sqlite)
else()
    find_package(SQLite3 3.24 REQUIRED)
    set(ROOTS_SQLITE_TARGET SQLite::SQLite3)
endif()

add_library(roots_tree SHARED src/roots_tree.cpp)
option(ROOTS_TREE_HOST_JNI "Build the JNI bridge on a desktop for contract checks" OFF)
if(ANDROID)
    target_sources(roots_tree PRIVATE src/roots_tree_jni.cpp)
elseif(ROOTS_TREE_HOST_JNI)
    find_package(JNI REQUIRED)
    target_sources(roots_tree PRIVATE src/roots_tree_jni.cpp)
    target_include_directories(roots_tree PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_include_directories(roots_tree PUBLIC include)
target_compile_definitions(roots_tree PRIVATE ROOTS_TREE_BUILD)
target_link_libraries(roots_tree PRIVATE ${ROOTS_SQLITE_TARGET})
if(ANDROID AND TARGET roots_tree_sqlite)
    # The app owns this SQLite instance; avoid symbol preemption by platform
    # SQLite or another library in the Android process.
    target_link_options(roots_tree PRIVATE "-Wl,--exclude-libs,libroots_tree_sqlite.a")
endif()
set_target_properties(roots_tree PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(roots_tree PRIVATE /W4 /permissive-)
else()
    target_compile_options(roots_tree PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(ROOTS_TREE_TESTS "Build native contract tests" ON)
if(ROOTS_TREE_TESTS AND NOT ANDROID)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(roots_tree_checks tests/tree_checks.cpp)
    target_link_libraries(roots_tree_checks PRIVATE roots_tree Threads::Threads)
    add_test(NAME roots_tree_checks COMMAND roots_tree_checks)
    add_executable(roots_work_checks tests/work_checks.cpp)
    target_link_libraries(roots_work_checks PRIVATE roots_tree)
    add_test(NAME roots_work_checks COMMAND roots_work_checks)
endif()
