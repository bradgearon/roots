cmake_minimum_required(VERSION 3.22.1)
project(roots_tree_core LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Android's platform sqlite is not a public NDK dependency. Provide an approved
# SQLite amalgamation for Android or a self-contained Windows build.
set(ROOTS_SQLITE_AMALGAMATION_DIR "" CACHE PATH "Directory containing sqlite3.c and sqlite3.h")
if(ROOTS_SQLITE_AMALGAMATION_DIR)
    add_library(roots_tree_sqlite STATIC "${ROOTS_SQLITE_AMALGAMATION_DIR}/sqlite3.c")
    target_include_directories(roots_tree_sqlite PUBLIC "${ROOTS_SQLITE_AMALGAMATION_DIR}")
    target_compile_definitions(roots_tree_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
    set_target_properties(roots_tree_sqlite PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(UNIX AND NOT ANDROID)
        find_package(Threads REQUIRED)
        target_link_libraries(roots_tree_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
    endif()
    set(ROOTS_SQLITE_TARGET roots_tree_sqlite)
else()
    if(ANDROID)
        message(FATAL_ERROR "Set ROOTS_SQLITE_AMALGAMATION_DIR to an approved SQLite amalgamation; do not link Android's private sqlite library.")
    endif()
    find_package(SQLite3 3.24 REQUIRED)
    set(ROOTS_SQLITE_TARGET SQLite::SQLite3)
endif()

add_library(roots_tree SHARED src/roots_tree.cpp)
if(ANDROID)
    target_sources(roots_tree PRIVATE src/roots_tree_jni.cpp)
endif()
target_include_directories(roots_tree PUBLIC include)
target_compile_definitions(roots_tree PRIVATE ROOTS_TREE_BUILD)
target_link_libraries(roots_tree PRIVATE ${ROOTS_SQLITE_TARGET})
set_target_properties(roots_tree PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(roots_tree PRIVATE /W4 /permissive-)
else()
    target_compile_options(roots_tree PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(ROOTS_TREE_TESTS "Build native contract tests" ON)
if(ROOTS_TREE_TESTS AND NOT ANDROID)
    enable_testing()
    find_package(Threads REQUIRED)
    add_executable(roots_tree_checks tests/tree_checks.cpp)
    target_link_libraries(roots_tree_checks PRIVATE roots_tree Threads::Threads)
    add_test(NAME roots_tree_checks COMMAND roots_tree_checks)
endif()
