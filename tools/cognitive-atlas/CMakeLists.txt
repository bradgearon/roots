cmake_minimum_required(VERSION 3.22)
project(roots_cognitive_atlas LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
# Set -DROOTS_LLAMA_CPP_DIR=... to reuse a local checkout at a known revision.
set(ROOTS_LLAMA_CPP_DIR "" CACHE PATH "Local llama.cpp source")
if(ROOTS_LLAMA_CPP_DIR)
  add_subdirectory("${ROOTS_LLAMA_CPP_DIR}" llama)
else()
  include(FetchContent)
  FetchContent_Declare(llama GIT_REPOSITORY https://github.com/ggml-org/llama.cpp.git GIT_TAG b11960 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(llama)
endif()
add_executable(atlas-score score.cpp)
target_link_libraries(atlas-score PRIVATE llama)
